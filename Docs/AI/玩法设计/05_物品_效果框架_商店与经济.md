### 1.4 稀有度抽取公式（★ 单次掷骰）

```gdscript
func get_tier_from_wave(wave) -> int:
    var rand = rand_range(0.0, 1.0)          # ★ 只掷一次
    var tier = Tier.COMMON
    var luck = Utils.get_stat("stat_luck") / 100.0
    for i in range(3, -1, -1):               # ★ 从 LEGENDARY 向下遍历
        var wave_base = max(0.0, ((wave - 1) - MIN_WAVE[i]) * WAVE_BONUS[i])
        var wave_chance = wave_base * (1 + luck)  if luck >= 0  else  wave_base / (1 + abs(luck))
        var chance = BASE[i] + wave_chance
        if rand <= min(chance, MAX[i]): tier = i; break
    return tier
```

$$P_i^{raw} = \min\Big(\text{BASE}_i + \max(0,\ (W-1)-\text{MINW}_i)\cdot \text{WB}_i \cdot f(L),\ \ \text{MAX}_i\Big),\quad f(L)=\begin{cases}1+L & L\ge 0\\ \frac{1}{1+|L|} & L<0\end{cases}$$

**零幸运时的阈值**：

| 波次 W | P(IV) | P(III) | P(II) |
|---|---|---|---|
| 1 | 0 | 0 | 0 |
| 3 | 0 | 0 | 0.12 |
| 5 | 0 | 0.04 | 0.24 |
| 7 | 0 | 0.08 | 0.36 |
| 10 | 0.0069 | 0.14 | 0.54 |
| 15 | 0.0184 | 0.24 | 0.60(封顶) |
| 20 | 0.0299 | 0.25(封顶) | 0.60 |
| 40 | 0.0759 | 0.25 | 0.60 |
| 41 | 0.0782 | 0.25 | 0.60 |
| **≥42** | **0.08(封顶)** | 0.25 | 0.60 |

> **★ 实现警告**：必须**单次 `rand` 复用 + 从高到低遍历**。若改成逐档独立掷骰，概率分布完全不同。
> 幸运只放大"波次增量项"，不影响 BASE，且受 MAX_CHANCE 硬封顶。Luck=+100% 时 W=10：P(IV)=0.0138，P(III)=0.25(封顶)，P(II)=0.60(封顶)。
> 零幸运时 P(IV) 直到 **W42** 才触及 0.08 上限（W41 = 0.0782），与 §02 §6.4 的升级稀有度表一致。

## 2. ★ Effect 系统架构（复刻核心）

### 2.1 基类（`items/global/effect.gd`）

```gdscript
class_name Effect extends Resource

enum Sign { POSITIVE, NEGATIVE, NEUTRAL, FROM_VALUE, FROM_ARG }
enum StorageMethod { SUM, KEY_VALUE, REPLACE }

export var key := ""                 # 目标 RunData.effects 键名
export var text_key := ""            # 本地化模板键（空则用 key.to_upper()）
export var value := 0
export var custom_key := ""          # 非空 → 写入 effects[custom_key] 数组
export var storage_method = SUM
export var effect_sign := FROM_VALUE
export var custom_args := []         # CustomArg：控制文本参数渲染
var base_value = 0                   # REPLACE 模式备份原值

static func get_id() -> String: return "effect"

func apply():
    if custom_key != "" or storage_method == KEY_VALUE:
        RunData.effects[custom_key].push_back([key, value])
    elif storage_method == REPLACE:
        base_value = RunData.effects[key]
        RunData.effects[key] = value
    else:
        RunData.effects[key] += value

func unapply():
    if custom_key != "" or storage_method == KEY_VALUE:
        RunData.effects[custom_key].erase([key, value])     # ★ 值比较，有 bug
    elif storage_method == REPLACE:
        RunData.effects[key] = base_value
    else:
        RunData.effects[key] -= value
```

> **★★★ 复刻必改点**：`Array.erase([key, value])` 依赖值相等比较，Godot 的 `erase` 只删第一个匹配项。**同一物品叠多份时移除会错删**。UE5 侧**必须用效果实例句柄（GUID / 索引）管理**，而不是值比较。

### 2.2 三种存储语义

| StorageMethod | 行为 | 用途 |
|---|---|---|
| `SUM` (0) | `effects[key] += value` | 99% 的普通属性加减 |
| `KEY_VALUE` (1) | `effects[custom_key].push_back([key, value])` | 需要保留"键+值"元组的复杂效果 |
| `REPLACE` (2) | 备份后覆写 | 上限类（`hp_cap` / `dodge_cap` / `weapon_slot`） |

> `custom_key != ""` 会**强制走数组分支**，即使 `storage_method == SUM`。这是最常用的复杂效果注入方式。

### 2.3 注册 / 结算流程（`run_data.gd`）

```gdscript
func add_item(item: ItemData):
    items.push_back(item)
    apply_item_effects(item)          # for effect in item.effects: effect.apply()
    add_item_displayed(item)          # 外观层叠（position / display_priority / depth 排序）
    update_item_related_effects()     # ★ 派生效果重算（顺序敏感）
    LinkedStats.reset()               # 联动属性全量重算
    check_bait_chal(item.my_id); check_scavenger_chal()
```

**`update_item_related_effects()` 的固定顺序**（`run_data.gd:497-552`）：
```
1. update_unique_bonuses()            # effects["unique_weapon_effects"] × 唯一 weapon_id 数
2. update_additional_weapon_bonuses() # effects["additional_weapon_effects"] × 武器总数
3. update_tier_iv_weapon_bonuses()    # effects["tier_iv_weapon_effects"] × Tier IV 武器数
4. update_tier_i_weapon_bonuses()     # effects["tier_i_weapon_effects"] × Tier I 武器数
```
每个都是 **"先全额回退上次累加 → 清空缓存列表 → 按当前武器重新累加"** 的幂等重算模式。

### 2.6 爆炸统一结算（`run_data.gd:829-846`）

```gdscript
func handle_explosion(key: String, pos: Vector2):
    if effects[key].size() > 0:
        var explosion_chance = 0.0
        for e in effects[key]: explosion_chance += e.chance        # ★ 概率线性相加（可 >100%）
        if Utils.get_chance_success(explosion_chance):
            var dmg = 0
            for e in effects[key]:
                dmg += WeaponService.init_base_stats(e.stats, "", [], [ExplodingEffect.new()]).damage
            WeaponService.explode(effects[key][0], pos, dmg, ...)   # 只生成 1 次，视觉用第 1 个的 scale
```
**叠加规则：概率相加、伤害相加，但只生成 1 次爆炸。**

### 2.7 文本渲染管线

`Effect.get_text(colored=true)`：
1. `key_text = text_key.empty() ? key.to_upper() : text_key.to_upper()`
2. `args = get_args()`（基类 = `[str(value), tr(key.to_upper())]`，各子类重写）
3. 每个 arg 初始 sign = `get_sign(effect_sign, value)`
4. 遍历 `custom_args`（`items/global/custom_arg.gd`）：按 `arg_index` 覆写，值来源 `ArgValue`，符号 `arg_sign`，格式 `Format`
5. `Text.text(key_text, args, signs)`

`get_arg_value()` 的动态来源：
| ArgValue | 计算 |
|---|---|
| `UNIQUE_WEAPONS` | `value × RunData.get_unique_weapon_ids().size()` |
| `ADDITIONAL_WEAPONS` | `value × RunData.weapons.size()` |
| `TIER_IV_WEAPONS` | `value × (tier >= LEGENDARY 的武器数)` |
| `TIER_I_WEAPONS` | `value × (tier <= COMMON 的武器数)` |
| `TIER` | `"TIER_I".."TIER_IV"` |
| `SCALING_STAT` | `Utils.get_scaling_stat_text(key, value/100)` |
| `SCALING_STAT_VALUE` | `WeaponService.get_scaling_stats_value([[key, value/100]])` |
| `MAX_NB_OF_WAVES` | `RunData.nb_of_waves` |

追踪文本：`ItemParentData.get_effects_text()` 末尾追加 `RunData.tracked_item_effects[my_id]`（27 个被追踪物品）。

### 2.8 序列化

每个子类实现 `serialize() -> Dictionary` / `deserialize_and_merge(Dictionary)`：


需要**完整序列化**的键（`run_data.gd:19-25`）：

```
effect_keys_full_serialization = ["burn_chance","structures","explode_on_hit","explode_on_death",
```

```
1. number = 4
2. 武器保底数：
     wave ≤ 2 → 2
     wave ≤ 5 → 1
     若 effects["minimum_weapons_in_shop"] 更大则取之
     已锁定的武器计入保底额
3. 每格类型判定：
     wave ≤ 2：未满保底 → 武器，否则 → 物品
     有 effects["guaranteed_shop_items"] 待放 → 强制物品
     否则：get_chance_success(0.35) 或未满保底 → 武器；else 物品
     effects["weapon_slot"] <= 0 → 强制全物品
4. excluded_items = 本次已生成 + 上次列表（防重复）
```

### 3.3 武器倾向（`get_rand_item_for_wave`）

单次 `rand_wanted = randf()`：
```

chance_same_weapon_set = 0.35 + bonus_same_set
```

| 区间 | 行为 |
|---|---|
| `rand < 0.20` | 池内**仅保留玩家已持有同 `weapon_id`** 的武器（若有） |
| `0.20 ≤ rand < chance_same_weapon_set` | 池内**仅保留与玩家武器共享职业** 的武器（若有） |
| 否则 | 不过滤 |

W=1 时 `chance_same_weapon_set = 0.50`；W≥6 时 = 0.35。

**Tier 钳制**：`item_tier = clamp(tier, effects["min_weapon_tier"], effects["max_weapon_tier"])`

### 3.5 ★ 定价公式（`item_service.gd:468-481`）

```gdscript
func get_value(wave, base_value, affected_by_items_price=true, is_weapon=false, item_id="") -> int:
    var v = base_value
    if is_weapon and affected_by_items_price:
        v = base_value * (1.0 + effects["weapons_price"] / 100.0)
    var specific = 0
    for p in effects["specific_items_price"]:
        if p[0] == item_id: specific = p[1]; break
    var items_price_factor = 1.0 + ((effects["items_price"] + specific) / 100.0)
    var diff_factor    = effects["inflation"] / 100.0
    var endless_factor = RunData.get_endless_factor(wave) / 5.0
    return max(1.0, (v + wave + v * wave * (0.1 + diff_factor))
                    * items_price_factor * (1 + endless_factor)) as int
```

$$\text{Price} = \max\!\left(1,\ \Big\lfloor \big(V' + W + V'\cdot W\cdot(0.1 + F)\big)\cdot\big(1+\tfrac{I_p}{100}\big)\cdot\big(1+\tfrac{E}{5}\big)\Big\rfloor\right)$$

> `as int` 是**截断向零**；因价格恒为正，此处等价于 `floor`。但 `recycling_value` / `hp_shop` 的 `ceil` 不可混用，见 §3.9 / §3.10。

**价格表（无加成，W ≤ 20 故 E=0）**：

| item.value | W=1 | W=5 | W=10 | W=15 | W=20 |
|---|---|---|---|---|---|
| 15（Tier I 典型） | 17 | 27 | 40 | 52 | 65 |
| 40（Tier II 典型） | 45 | 65 | 90 | 115 | 140 |
| 70（Tier III 典型） | 78 | 110 | 150 | 190 | 230 |
| 100（Tier IV 典型） | 111 | 155 | 210 | 265 | 320 |
| 120（最贵档） | 133 | 185 | 250 | 315 | 380 |

### 3.6 重掷价格（累进）

```gdscript
func get_reroll_price(wave, last_reroll_value) -> int:
    return max(1.0, last_reroll_value + max(1.0, (0.5 * wave * pow(1.0 + endless_factor, 0.5)))) as int
```
$$\text{Reroll}_{n+1} = \max\Big(1,\ \text{Reroll}_n + \max\big(1,\ 0.5\cdot W\cdot\sqrt{1+E}\big)\Big)$$

初始化：`_last_reroll_price = get_reroll_price(current_wave, current_wave)` → 即 `W + max(1, ⌊0.5W⌋)`

**递进**（`shop.gd:254-266`）：
```gdscript

```

| +100% | 2.0% | 2.0% |

### 6.2 结构物伤害公式

```gdscript
# is_structure = true 时：
atk_spd = 0                              # 不吃攻速
damage = max(1, base.damage + get_scaling_stats_value(scaling_stats))
percent_dmg_bonus = 1                    # ★ 不吃 stat_percent_damage
if is_exploding: exploding_dmg_bonus = explosion_damage / 100
damage = max(1, round(damage * (1 + exploding_dmg_bonus)))
crit_chance = base.crit_chance           # 不吃 stat_crit_chance
lifesteal   = base.lifesteal             # 不吃 stat_lifesteal
max_range   = base.max_range             # 不吃 stat_range
```

$$D_{structure} = \max\Big(1,\ \text{round}\big((D_{base} + \textstyle\sum \text{Stat}_i k_i)\cdot(1 + \tfrac{\text{explosion\_damage}}{100})\big)\Big)$$

**示例**：标准炮塔 + 工程 20 → `10 + 20×0.8 = 26` 伤害，每 0.53 s 一发
火箭炮塔 + 工程 20 + 爆炸伤害 60 → `(25 + 30) × 1.6 = 88` 伤害

炮塔射击间隔随机化：`_cooldown = rand_range(max(1, cd×0.7), cd×1.3)`

### 6.3 生成逻辑（`global/entity_spawner.gd:337-371`）

```gdscript
cur_time = (wave_timer.wait_time - wave_timer.time_left) as int
spawn_radius = min(600, 400 + effects["structures"].size() * 10)
base_pos = ZoneService.get_rand_pos(600 + 64)

for struct in effects["structures"]:
    spawn_cd = struct.spawn_cooldown
    if spawn_cd != -1 and structures_cooldown_reduction 存在:
        spawn_cd = Utils.apply_cooldown_reduction(spawn_cd, reduction)
    if (spawn_cd != -1 and cur_time % spawn_cd == 0) or spawn_all:    # spawn_all = 波首
        for nb in struct.value:
            pos = group_structures && can_be_grouped
                  ? get_spawn_pos_in_area(base_pos, spawn_radius)
                  : ZoneService.get_rand_pos(160)
            if struct.spawn_around_player != -1:
                pos = get_spawn_pos_in_area(player.pos, struct.spawn_around_player)
            queue_to_spawn_structures.push_back(...)
```
- `spawn_cooldown == -1` → **仅波首生成一次**
- `group_structures`（Engineer）→ 所有可分组结构物集中在半径 `min(600, 400+N×10)` 内
- `tree_turrets`（口袋工厂）：树死亡时在其周围 200 px 生成炮塔
- `MAX_STRUCTURES = 100`

**冷却缩减**：`new_cd = max(2, base_cd × (1/(1+total)))`（total>0）或 `max(2, base_cd × (1+|total|))`

## 7. 状态效果数值

### 7.1 ★ 燃烧（Burning）

**数据结构**：
```gdscript
class BurningData:
    chance   : float = 0.0     # 0–1
    damage   : int = 0         # 每 tick 伤害
    duration : int = 0         # ★ tick 数，不是秒！
    spread   : int = 0
    type     : {ELEMENTAL=0, ENGINEERING=1}
    from     : Node            # 伤害归属
    merge(d) / remove(d)       # 各字段加/减
```

**tick 周期**：
$$T_{tick} = \max\big(0.1,\ 0.5\cdot(1 - \tfrac{\text{burning\_cooldown\_reduction}}{100})\big)\ \text{秒}$$
（`unit.tscn` 的 `BurningTimer.wait_time = 0.5`）

**伤害计算**（`weapon_service.gd:149-174`）：
```gdscript
if 武器本身 chance == 0:      # 完全继承全局
    new.chance = global.chance; new.damage = global.damage; new.duration = global.duration
elif 武器自带燃烧:             # 只叠加全局伤害
    new.damage += global.damage
new.spread += effects["burning_spread"]
if not is_structure or (is_structure and 武器无燃烧):
    new.damage += Utils.get_stat("stat_elemental_damage")
    new.damage = max(1, round(new.damage * (1 + stat_percent_damage/100)))
```

$$D_{tick} = \max\Big(1,\ \text{round}\big((D_{base} + D_{global} + \text{ELE})\cdot(1+\tfrac{\%dmg}{100})\big)\Big)$$
$$\text{TotalDamage} = D_{tick}\times \text{duration},\qquad \text{TotalTime} = \text{duration}\times T_{tick}$$

**施加与叠加（★ 取最大值刷新，不叠层）**：
```gdscript
if _burning != null:
    if _burning.damage < new.damage: _burning.type = new.type; _burning.from = new.from
    _burning.chance   = max(_burning.chance,   new.chance)
    _burning.damage   = max(_burning.damage,   new.damage)
    _burning.duration = max(_burning.duration, new.duration)
    _burning.spread   = max(_burning.spread,   new.spread)
else:
    _burning = new.duplicate(); _burning_timer.start()
```

**tick 结算（★ 无视护甲与闪避）**：
```gdscript
dmg = take_damage(_burning.damage, null, dodgeable=false, armor_applied=false, burn_sound, 0.1)
_burning.duration -= 1
if _burning.duration <= 0: _burning = null; 停表; 关粒子
```

### 8.3 验收用例

